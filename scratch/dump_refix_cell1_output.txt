def calculate_daily_spread(stock_pairs, data):
    spread = pd.DataFrame(index=data.index)
    for _, row in stock_pairs.iterrows():
        s1, s2 = row['Stock1'], row['Stock2']
        pair = f"{s1}-{s2}"
        if s1 in data.columns and s2 in data.columns:
            spread[pair] = data[s1] - data[s2]
    return spread


def generate_trade_signals(price_diff, mean_spread, std_spread):
    upper = mean_spread + 2 * std_spread
    lower = mean_spread - 2 * std_spread
    df = pd.DataFrame(index=price_diff.index)
    df['spread'] = price_diff
    df['spread_prev'] = df['spread'].shift(1)
    df['signal1'] = 0
    df['position1'] = 0

    for t in df.index[1:]:
        prev = df.at[t, 'spread_prev']
        curr = df.at[t, 'spread']
        pos_prev = df.at[df.index[df.index.get_loc(t) - 1], 'position1']
        if pos_prev == 0:
            if prev > upper and curr <= upper:
                df.at[t, 'signal1'] = -1
            elif prev < lower and curr >= lower:
                df.at[t, 'signal1'] = 1
        else:
            if pos_prev == -1 and curr >= mean_spread:
                df.at[t, 'signal1'] = 1
            elif pos_prev == 1 and curr <= mean_spread:
                df.at[t, 'signal1'] = -1
        df.at[t, 'position1'] = pos_prev + df.at[t, 'signal1']

    penult = df.index[-2]
    last = df.index[-1]
    if df.at[penult, 'position1'] != 0:
        df.at[last, 'signal1'] = -df.at[penult, 'position1']
        df.at[last, 'position1'] = 0

    df['signal2'] = -df['signal1']
    df['position2'] = -df['position1']
    df['trade1'] = df['signal1']
    df['trade2'] = df['signal2']
    return df[['spread', 'position1', 'position2', 'trade1', 'trade2']]


def generate_trade_records(signals, trade_data, initial_capital=100000, stop_loss_pct=0.10):
    half_cap = initial_capital / 2
    records = []
    for pair in signals['pair'].unique():
        sig = signals[signals['pair'] == pair]
        s1, s2 = pair.split('-')
        p1 = trade_data[s1]; p2 = trade_data[s2]
        rec = pd.DataFrame(index=sig.index)
        rec['trade1'] = sig['trade1']
        rec['trade2'] = sig['trade2']
        rec['price1'] = p1
        rec['price2'] = p2
        rec['cash1'] = np.nan; rec['cash2'] = np.nan
        rec['holdings1'] = np.nan; rec['holdings2'] = np.nan
        rec['total_value'] = np.nan
        rec['stop_loss'] = 0

        cash1 = cash2 = half_cap
        hold1 = hold2 = 0.0
        rec.iloc[0, rec.columns.get_loc('cash1')] = cash1
        rec.iloc[0, rec.columns.get_loc('cash2')] = cash2
        rec.iloc[0, rec.columns.get_loc('holdings1')] = hold1
        rec.iloc[0, rec.columns.get_loc('holdings2')] = hold2
        rec.iloc[0, rec.columns.get_loc('total_value')] = cash1 + cash2

        for i in range(1, len(rec)):
            t = rec.index[i]
            t1 = rec.at[t, 'trade1']; t2 = rec.at[t, 'trade2']
            pr1 = rec.at[t, 'price1']; pr2 = rec.at[t, 'price2']
            if t1 != 0:
                cash1 += -t1 * half_cap - abs(half_cap) * 0.001
                hold1 = t1 * (half_cap / pr1)
            if t2 != 0:
                cash2 += -t2 * half_cap - abs(half_cap) * 0.001
                hold2 = t2 * (half_cap / pr2)
            total = cash1 + cash2 + hold1 * pr1 + hold2 * pr2
            loss_pct = (initial_capital - total) / initial_capital
            if loss_pct >= stop_loss_pct and (hold1 != 0 or hold2 != 0):
                cash1 += hold1 * pr1; cash2 += hold2 * pr2
                hold1 = hold2 = 0.0
                rec.at[t, 'stop_loss'] = 1
                total = cash1 + cash2
            else:
                rec.at[t, 'stop_loss'] = 0
            rec.at[t, 'cash1'] = cash1; rec.at[t, 'cash2'] = cash2
            rec.at[t, 'holdings1'] = hold1; rec.at[t, 'holdings2'] = hold2
            rec.at[t, 'total_value'] = total
        rec['pair'] = pair
        records.append(rec)
    return pd.concat(records)

def compute_nav_series(total_value_series, initial_capital=100000):
    """將 total_value 轉為淨值（以1為基準）"""
    return total_value_series / initial_capital


def compute_max_drawdown(nav_series):
    """最大回撤"""
    rolling_max = nav_series.cummax()
    drawdown = (nav_series - rolling_max) / rolling_max
    return drawdown.min(), drawdown


def compute_sharpe(daily_returns, risk_free_rate=0.02, trading_days=252):
    """年化夏普比率"""
    excess = daily_returns - risk_free_rate / trading_days
    if excess.std() == 0:
        return np.nan
    return np.sqrt(trading_days) * excess.mean() / excess.std()


def compute_calmar(annualized_return, max_drawdown):
    """卡瑪比率"""
    if max_drawdown == 0:
        return np.nan
    return annualized_return / abs(max_drawdown)


def compute_sortino(daily_returns, risk_free_rate=0.02, trading_days=252):
    """索提諾比率"""
    excess = daily_returns - risk_free_rate / trading_days
    downside = excess[excess < 0]
    if len(downside) == 0 or downside.std() == 0:
        return np.nan
    return np.sqrt(trading_days) * excess.mean() / downside.std()


def compute_win_rate(trade_pnl_list):
    """勝率"""
    if len(trade_pnl_list) == 0:
        return np.nan
    wins = sum(1 for p in trade_pnl_list if p > 0)
    return wins / len(trade_pnl_list)


def compute_profit_factor(trade_pnl_list):
    """獲利因子"""
    gains = sum(p for p in trade_pnl_list if p > 0)
    losses = abs(sum(p for p in trade_pnl_list if p < 0))
    if losses == 0:
        return np.inf
    return gains / losses


def extract_trade_pnl(rec_df, initial_capital=100000):
    """從交易記錄提取每筆交易盈虧"""
    pnl = []
    half = initial_capital / 2
    in_trade = False
    entry_val = initial_capital
    for i, row in rec_df.iterrows():
        if (row['trade1'] != 0 or row['trade2'] != 0) and not in_trade:
            entry_val = row['total_value']
            in_trade = True
        elif (row['trade1'] != 0 or row['trade2'] != 0) and in_trade:
            pnl.append(row['total_value'] - entry_val)
            entry_val = row['total_value']
    return pnl


def full_performance_report(tr_records, initial_capital=100000, risk_free=0.02):
    """計算所有配對的完整績效報告"""
    results = []
    for pair, grp in tr_records.groupby('pair'):
        grp = grp.sort_index()
        tv = grp['total_value'].ffill()

        nav = compute_nav_series(tv, initial_capital)
        daily_ret = tv.pct_change().dropna()
        n_days = len(grp)
        years = n_days / 252

        total_ret = (tv.iloc[-1] - initial_capital) / initial_capital
        ann_ret = (1 + total_ret) ** (1 / years) - 1 if years > 0 else 0
        mdd_val, _ = compute_max_drawdown(nav)
        sharpe = compute_sharpe(daily_ret, risk_free)
        sortino = compute_sortino(daily_ret, risk_free)
        calmar = compute_calmar(ann_ret, mdd_val)
        pnl_list = extract_trade_pnl(grp, initial_capital)
        win_rate = compute_win_rate(pnl_list)
        profit_factor = compute_profit_factor(pnl_list)
        n_trades = len(pnl_list)

        results.append({
            '配對': pair,
            '總報酬率(%)': round(total_ret * 100, 2),
            '年化報酬率(%)': round(ann_ret * 100, 2),
            '最大回撤(%)': round(mdd_val * 100, 2),
            '夏普比率': round(sharpe, 3) if not np.isnan(sharpe) else 'N/A',
            '索提諾比率': round(sortino, 3) if not np.isnan(sortino) else 'N/A',
            '卡瑪比率': round(calmar, 3) if not np.isnan(calmar) else 'N/A',
            '勝率(%)': round(win_rate * 100, 1) if not np.isnan(win_rate) else 'N/A',
            '獲利因子': round(profit_factor, 3) if profit_factor != np.inf else '∞',
            '交易次數': n_trades,
        })
    return pd.DataFrame(results)

# ─────────────────────────────────────────────
#  9. 前20組：價格走勢 + 進出場點位
# ─────────────────────────────────────────────

def plot_pair_entry_exit(tr_records, trade_data, stock_pairs,
                         spread_t, mean_spread, std_spread,
                         round_label='最後一輪', initial_capital=100000):
    """
    為每個配對繪製：
      上方：兩支股票的標準化價格走勢 + 進出場箭頭
      下方：價差走勢 + 上下軌 + 均值
    """
    pairs = stock_pairs.apply(lambda x: f"{x['Stock1']}-{x['Stock2']}", axis=1).tolist()
    n = len(pairs)
    cols = 2
    rows = n  # 每個pair佔一大行，每大行分上下子圖

    fig = plt.figure(figsize=(22, rows * 5), facecolor=STYLE['bg'])
    fig.suptitle(f'配對交易 ─ {round_label} 前{n}對：價格走勢與進出場點位',
                 fontsize=15, color=STYLE['text'], fontweight='bold', y=1.001)

    outer = gridspec.GridSpec(rows, cols, figure=fig,
                              hspace=0.55, wspace=0.35,
                              top=0.995, bottom=0.02,
                              left=0.05, right=0.97)

    for i, pair in enumerate(pairs):
        row_i = i // cols
        col_i = i % cols
        inner = gridspec.GridSpecFromSubplotSpec(
            2, 1, subplot_spec=outer[row_i, col_i],
            height_ratios=[2, 1], hspace=0.05)

        ax_price = fig.add_subplot(inner[0])
        ax_spread = fig.add_subplot(inner[1], sharex=ax_price)

        s1, s2 = pair.split('-')

        # ── 標準化價格 ──────────────────────────
        p1 = trade_data[s1] / trade_data[s1].iloc[0]
        p2 = trade_data[s2] / trade_data[s2].iloc[0]

        styled_ax(ax_price, title=f'{pair}')
        ax_price.plot(p1.index, p1.values, color=STYLE['blue'],
                      linewidth=1.2, label=s1, alpha=0.9)
        ax_price.plot(p2.index, p2.values, color=STYLE['orange'],
                      linewidth=1.2, label=s2, alpha=0.9)

        # ── 進出場箭頭 ──────────────────────────
        rec = tr_records[tr_records['pair'] == pair]
        entry_up   = rec[(rec['trade1'] == 1)  & (rec['position1'].shift(1).fillna(0) == 0)]
        entry_down = rec[(rec['trade1'] == -1) & (rec['position1'].shift(1).fillna(0) == 0)]
        exit_dates = rec[(rec['trade1'] != 0)  & (rec['position1'] == 0)]

        y_range = max(p1.max(), p2.max()) - min(p1.min(), p2.min())
        arrow_h = y_range * 0.06

        for dt in entry_up.index:
            y = min(p1.loc[dt] if dt in p1.index else p1.iloc[-1],
                    p2.loc[dt] if dt in p2.index else p2.iloc[-1])
            ax_price.annotate('', xy=(dt, y + arrow_h), xytext=(dt, y),
                              arrowprops=dict(arrowstyle='->', color=STYLE['green'],
                                             lw=1.5))
        for dt in entry_down.index:
            y = max(p1.loc[dt] if dt in p1.index else p1.iloc[0],
                    p2.loc[dt] if dt in p2.index else p2.iloc[0])
            ax_price.annotate('', xy=(dt, y - arrow_h), xytext=(dt, y),
                              arrowprops=dict(arrowstyle='->', color=STYLE['red'],
                                             lw=1.5))
        for dt in exit_dates.index:
            y = (p1.loc[dt] + p2.loc[dt]) / 2 if dt in p1.index else p1.mean()
            ax_price.axvline(dt, color=STYLE['yellow'], linewidth=0.8,
                             linestyle=':', alpha=0.7)

        ax_price.legend(fontsize=7, facecolor=STYLE['panel'],
                        labelcolor=STYLE['text'], edgecolor=STYLE['border'],
                        loc='upper left')
        ax_price.set_ylabel('標準化價格', fontsize=7, color=STYLE['subtext'])
        plt.setp(ax_price.get_xticklabels(), visible=False)

        # ── 價差走勢 ────────────────────────────
        styled_ax(ax_spread, ylabel='價差')
        sp = spread_t[pair]
        mu = mean_spread[pair]
        sd = std_spread[pair]
        upper = mu + 2 * sd
        lower = mu - 2 * sd

        ax_spread.plot(sp.index, sp.values, color=STYLE['purple'],
                       linewidth=1, alpha=0.9, label='價差')
        ax_spread.axhline(mu,    color=STYLE['text'],   linewidth=0.8,
                          linestyle='--', alpha=0.6, label='均值')
        ax_spread.axhline(upper, color=STYLE['red'],    linewidth=0.8,
                          linestyle=':', alpha=0.7, label='+2σ')
        ax_spread.axhline(lower, color=STYLE['green'],  linewidth=0.8,
                          linestyle=':', alpha=0.7, label='-2σ')
        ax_spread.fill_between(sp.index, lower, upper,
                               alpha=0.07, color=STYLE['purple'])
        ax_spread.legend(fontsize=6, facecolor=STYLE['panel'],
                         labelcolor=STYLE['text'], edgecolor=STYLE['border'],
                         ncol=4, loc='upper right')
        ax_spread.xaxis.set_major_formatter(mdates.DateFormatter('%Y-%m'))
        ax_spread.xaxis.set_major_locator(mdates.AutoDateLocator())
        plt.setp(ax_spread.get_xticklabels(), rotation=25, fontsize=6)

    plt.savefig('pair_entry_exit.png', dpi=130, bbox_inches='tight',
                facecolor=STYLE['bg'])
    plt.show()
    print("✅ 進出場圖已儲存：pair_entry_exit.png")
